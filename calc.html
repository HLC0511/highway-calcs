// =========密码校验=========
const correctPwd = "HLC";
function checkPassword(){
    const val = document.getElementById("pwdInput").value.trim();
    if(val === correctPwd){
        document.getElementById("pwdMask").style.display = "none";
        document.getElementById("mainPage").style.display = "block";
        initApp();
    }else{
        alert("密码错误，请重新输入！");
        document.getElementById("pwdInput").value = "";
        document.getElementById("pwdInput").focus();
    }
}
//回车触发确认
document.getElementById("pwdInput").addEventListener("keydown",function(e){
    if(e.key === "Enter") checkPassword();
});

function initApp(){
    const laneRange = {
        4: [25000, 55000],
        6: [45000, 80000],
        8: [60000, 100000]
    };

    const servFactor = {
        "4_25000": [0.60,0.85,1.00,1.15,1.25],
        "4_40000": [0.64,0.90,1.09,1.25,1.35],
        "6_45000": [0.59,0.85,1.00,1.12,1.25],
        "6_60000": [0.73,0.99,1.20,1.38,1.51],
        "8_60000": [0.59,0.82,1.00,1.14,1.24],
        "8_80000": [0.65,0.93,1.09,1.24,1.36]
    };
    const parkFactor = {
        "4_25000": [0.81,0.92,1.00,1.08,1.16],
        "4_40000": [1.01,1.11,1.20,1.30,1.39],
        "6_45000": [0.82,0.91,1.00,1.09,1.16],
        "6_60000": [0.97,1.04,1.12,1.19,1.25],
        "8_60000": [0.87,0.93,1.00,1.06,1.10],
        "8_80000": [0.92,1.02,1.11,1.19,1.26]
    };
    const baseServ = { land:113, buildArea:6020.64, bigP:312, midP:74, smallP:74, toiletTotal:68, toiletM:26, toiletW:42 };
    const basePark = { land:27, buildArea:1438.56, bigP:36, midP:10, smallP:32, toiletTotal:12, toiletM:4, toiletW:8 };

    const qInput = document.getElementById("Qaadt");
    const laneSel = document.getElementById("lane");
    const muInput = document.getElementById("mu");
    const qRangeTip = document.getElementById("qRangeTip");

    function updateValidate(){
        const lane = Number(laneSel.value);
        const Q = Number(qInput.value);
        const mu = Number(muInput.value);
        const [rMin,rMax] = laneRange[lane];
        qRangeTip.innerText = `当前车道查表区间：${rMin}‑${rMax}`;

        if(isNaN(Q) || Q<rMin || Q>rMax){
            qInput.classList.add("input-warn");
        }else{
            qInput.classList.remove("input-warn");
        }
        if(isNaN(mu)||mu<0||mu>100){
            muInput.classList.add("input-warn");
        }else{
            muInput.classList.remove("input-warn");
        }
    }

    function getMuIdx(mu){
        if(mu<10) return 0;
        else if(mu<=20) return 1;
        else if(mu<=30) return 2;
        else if(mu<=40) return 3;
        else return 4;
    }

    function getFactorKey(lane,Q){
        if(lane===4){
            if(Q>=25000 && Q<=40000) return "4_25000";
            else if(Q>40000 && Q<=55000) return "4_40000";
        }else if(lane===6){
            if(Q>=45000 && Q<=60000) return "6_45000";
            else if(Q>60000 && Q<=80000) return "6_60000";
        }else if(lane===8){
            if(Q>=60000 && Q<=80000) return "8_60000";
            else if(Q>80000 && Q<=100000) return "8_80000";
        }
        return null;
    }

    function calc(){
        updateValidate();
        const Q = Number(qInput.value);
        const lane = Number(laneSel.value);
        const mu = Number(muInput.value);
        const warnDom = document.getElementById("warning");
        warnDom.innerText="";
        const [rMin,rMax] = laneRange[lane];

        const fKey = getFactorKey(lane,Q);
        if(!fKey){
            warnDom.innerText=`⚠️警告：当前${lane}车道规范查表区间${rMin}～${rMax}，输入交通量${Q}不在查表区间内，计算结果仅供参考！`;
        }
        if(mu<0 || mu>100){
            warnDom.innerText += " ⚠️大型车比例μ应当在0‑100之间！";
        }

        const muIdx = getMuIdx(mu);
        let servCorr = 1;
        let parkCorr = 1;
        if(fKey){
            servCorr = servFactor[fKey][muIdx];
            parkCorr = parkFactor[fKey][muIdx];
        }

        const serv = {
            land: Math.round(baseServ.land * servCorr),
            buildArea: Math.round(baseServ.buildArea * servCorr*100)/100,
            bigP: Math.round(baseServ.bigP * servCorr),
            midP: Math.round(baseServ.midP * servCorr),
            smallP: Math.round(baseServ.smallP * servCorr),
            toiletTotal: Math.round(baseServ.toiletTotal * servCorr),
            toiletM: Math.round(baseServ.toiletM * servCorr),
            toiletW: Math.round(baseServ.toiletW * servCorr)
        };
        serv.chargeP = Math.round((serv.bigP+serv.midP+serv.smallP)*0.20);

        const park = {
            land: Math.round(basePark.land * parkCorr),
            buildArea: Math.round(basePark.buildArea * parkCorr*100)/100,
            bigP: Math.round(basePark.bigP * parkCorr),
            midP: Math.round(basePark.midP * parkCorr),
            smallP: Math.round(basePark.smallP * parkCorr),
            toiletTotal: Math.round(basePark.toiletTotal * parkCorr),
            toiletM: Math.round(basePark.toiletM * parkCorr),
            toiletW: Math.round(basePark.toiletW * parkCorr)
        };
        park.chargeP = Math.round((park.bigP+park.midP+park.smallP)*0.20);

        const sTbody = document.getElementById("servTable");
        sTbody.innerHTML = `
        <tr><td>用地规模(亩)</td><td>${serv.land}</td><td>${(serv.land/2).toFixed(1)}</td></tr>
        <tr><td>建筑最大占地面积(㎡)</td><td>${serv.buildArea}</td><td>${(serv.buildArea/2).toFixed(2)}</td></tr>
        <tr><td>大型车停车位</td><td>${serv.bigP}</td><td>${serv.bigP/2}</td></tr>
        <tr><td>中型车停车位</td><td>${serv.midP}</td><td>${serv.midP/2}</td></tr>
        <tr><td>小型车停车位</td><td>${serv.smallP}</td><td>${serv.smallP/2}</td></tr>
        <tr><td>充电车位(20%)</td><td>${serv.chargeP}</td><td>${serv.chargeP/2}</td></tr>
        <tr><td>公厕总厕位</td><td>${serv.toiletTotal}</td><td>${serv.toiletTotal/2}</td></tr>
        <tr><td>男厕位</td><td>${serv.toiletM}</td><td>${serv.toiletM/2}</td></tr>
        <tr><td>女厕位</td><td>${serv.toiletW}</td><td>${serv.toiletW/2}</td></tr>
        `;
        const pTbody = document.getElementById("parkTable");
        pTbody.innerHTML = `
        <tr><td>用地规模(亩)</td><td>${park.land}</td><td>${(park.land/2).toFixed(1)}</td></tr>
        <tr><td>建筑最大占地面积(㎡)</td><td>${park.buildArea}</td><td>${(park.buildArea/2).toFixed(2)}</td></tr>
        <tr><td>大型车停车位</td><td>${park.bigP}</td><td>${park.bigP/2}</td></tr>
        <tr><td>中型车停车位</td><td>${park.midP}</td><td>${park.midP/2}</td></tr>
        <tr><td>小型车停车位</td><td>${park.smallP}</td><td>${park.smallP/2}</td></tr>
        <tr><td>充电车位(20%)</td><td>${park.chargeP}</td><td>${park.chargeP/2}</td></tr>
        <tr><td>公厕总厕位</td><td>${park.toiletTotal}</td><td>${park.toiletTotal/2}</td></tr>
        <tr><td>男厕位</td><td>${park.toiletM}</td><td>${park.toiletM/2}</td></tr>
        <tr><td>女厕位</td><td>${park.toiletW}</td><td>${park.toiletW/2}</td></tr>
        `;
    }

    laneSel.addEventListener("change",()=>{updateValidate();calc();});
    qInput.addEventListener("input",updateValidate);
    muInput.addEventListener("input",updateValidate);
    document.getElementById("btnCalc").addEventListener("click", calc);

    updateValidate();
    calc();
}
</script>
